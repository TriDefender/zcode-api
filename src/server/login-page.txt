(function () {
  "use strict";
  var KEY_STORAGE = "zcode-login-key";
  var LANG_STORAGE = "zcode-login-lang";
  var key = sessionStorage.getItem(KEY_STORAGE) || "";
  // Language: manual override (localStorage) wins, else navigator.language
  // with zh* → Chinese and everything else → English.
  var lang = localStorage.getItem(LANG_STORAGE) ||
    ((navigator.language || "").toLowerCase().indexOf("zh") === 0 ? "zh" : "en");
  var statusEl = document.getElementById("status");
  var cards = Array.prototype.slice.call(document.querySelectorAll(".card"));
  var pollTimer = null;

  var T = {
    zh: {
      title: "ZCode Proxy 登录",
      h1: "ZCode Proxy 登录",
      intro: "选择要登录的站点。授权页由服务端下发（国际站 chat.z.ai / 中国站 zchat.z.ai），会在当前标签页内打开；在浏览器里完成登录后按浏览器「返回」（可能需按两次）回到本页即可看到结果——无需本地回调。BigModel / 智谱账号请使用「中国版」登录（账号已互链，订阅自动识别）。",
      cardZaiName: "Z.ai 国际版",
      cardZaiDesc: "国际站账号 · 国内网络会经区域跳转到达中国站登录页",
      cardZaiCnName: "Z.ai 中国版",
      cardZaiCnDesc: "中国站账号（含 BigModel/智谱订阅） · 直达中国站登录页，无需中转",
      connecting: "正在连接…",
      selectPrompt: "请选择上方站点开始登录",
      fetching: "正在获取授权地址…",
      jumping: "正在跳转授权页…",
      startFail: "启动登录失败：",
      retryHint: "（重试请刷新页面）",
      alreadyInProgress: "已有进行中的登录——完成授权后按「返回」回到本页查看结果",
      waitAuth: "等待授权完成…（在浏览器里完成站点登录即可）",
      ready: "✅ 登录成功，可以关闭此页",
      failed: "❌ 登录失败：",
      refreshHint: "（刷新页面可重试）",
      retired: "登录服务已结束。若已完成登录，本页可以关闭；否则请重新发起登录。",
      keyPrompt: "此端点需要 Proxy API Key（config.yaml 中的 proxyApiKey）：",
      keyWrong: "unauthorized（API Key 不正确）",
      backHere: "完成授权后按「返回」回到本页查看结果"
    },
    en: {
      title: "ZCode Proxy Login",
      h1: "ZCode Proxy Login",
      intro: "Pick a site to log in. The authorize page is issued by the server (international chat.z.ai / China zchat.z.ai) and opens in this tab; after signing in, use the browser BACK button (possibly twice) to return here — no local callback needed. For BigModel/智谱 subscriptions use \"Z.ai China\" (accounts are linked; your plan is recognized automatically).",
      cardZaiName: "Z.ai International",
      cardZaiDesc: "International account · from CN networks it region-bounces to the China login page",
      cardZaiCnName: "Z.ai China",
      cardZaiCnDesc: "China account (incl. BigModel/智谱 plans) · straight to the China login page, no detour",
      connecting: "Connecting…",
      selectPrompt: "Pick a site above to start",
      fetching: "Requesting the authorize URL…",
      jumping: "Redirecting to the authorize page…",
      startFail: "Login failed to start: ",
      retryHint: " (refresh to retry)",
      alreadyInProgress: "A login is already in progress — finish authorizing, then press BACK to return here",
      waitAuth: "Waiting for authorization… (finish signing in in the browser)",
      ready: "✅ Login complete — you can close this page",
      failed: "❌ Login failed: ",
      refreshHint: " (refresh to retry)",
      retired: "The login service has ended. If you already finished, you can close this page; otherwise start a new login.",
      keyPrompt: "This endpoint needs the Proxy API Key (proxyApiKey in config.yaml):",
      keyWrong: "unauthorized (wrong API key)",
      backHere: "After authorizing, press BACK to return here for the result"
    }
  };
  function t(key) {
    return (T[lang] && T[lang][key]) || T.zh[key] || key;
  }

  function authHeaders() {
    return key ? { "authorization": "Bearer " + key } : {};
  }
  function setStatus(text, kind) {
    statusEl.textContent = text;
    statusEl.className = "status" + (kind ? " " + kind : "");
  }
  function setCardsDisabled(disabled) {
    cards.forEach(function (c) { c.disabled = disabled; });
  }
  // api(): shared fetch wrapper — on 401 (main-server proxyApiKey gate) prompt
  // once for the key, remember it in sessionStorage, and retry. Fetch
  // rejections (listener closed after the completion grace) come back as an
  // in-band error so the polling loop can degrade gracefully.
  async function api(path, opts) {
    var r;
    try {
      r = await fetch(path, opts);
    } catch (e) {
      return { ok: false, error: "network" };
    }
    if (r.status === 401 && !key) {
      var k = prompt(t("keyPrompt"));
      if (k && k.trim()) {
        key = k.trim();
        sessionStorage.setItem(KEY_STORAGE, key);
        var retryOpts = Object.assign({}, opts, { headers: Object.assign({}, (opts && opts.headers) || {}, authHeaders()) });
        r = await fetch(path, retryOpts);
      }
    }
    var parsed = null;
    try { parsed = await r.json(); } catch (e) { /* non-JSON */ }
    if (!parsed || typeof parsed.ok !== "boolean") {
      return { ok: false, error: r.status === 401 ? t("keyWrong") : "HTTP " + r.status };
    }
    return parsed;
  }

  async function start(provider) {
    setCardsDisabled(true);
    setStatus(t("fetching"));
    var r = await api("/login/start", {
      method: "POST",
      headers: Object.assign({ "content-type": "application/json" }, authHeaders()),
      body: JSON.stringify({ provider: provider })
    });
    if (!r.ok) {
      // A flow left authorizing from an earlier navigation (same-tab jump +
      // Back) is not an error — resume its status display instead.
      if (r.error === "login_already_in_progress") {
        setStatus(t("alreadyInProgress"));
        startPolling();
        return;
      }
      setStatus(t("startFail") + (r.error || "unknown") + t("retryHint"), "err");
      setCardsDisabled(false);
      return;
    }
    // SAME-TAB navigation (mobile-friendly: no popups, no tab juggling —
    // decision 2026-11-14). The OAuth page takes over this tab; the login
    // flow completes server-side meanwhile. After authorizing, the browser
    // lands on the interstitial (the zcode:// deep-link attempt fails
    // silently in a plain browser) — pressing BACK re-loads this page and
    // the on-load pollOnce() resumes the status display.
    setStatus(t("jumping"));
    window.location.href = r.authorizeUrl;
  }

  function stopPolling() {
    if (pollTimer !== null) { clearInterval(pollTimer); pollTimer = null; }
  }
  function startPolling() {
    stopPolling();
    pollOnce();
    pollTimer = setInterval(pollOnce, 2000);
  }
  var pollInFlight = false;
  async function pollOnce() {
    if (pollInFlight) return;
    pollInFlight = true;
    try {
      var r = await api("/login/status", { headers: authHeaders() });
      if (r.error === "network") {
        // The login listener is gone: completion grace elapsed (or the TUI
        // closed it). Nothing left to poll.
        setStatus(t("retired"), "err");
        stopPolling();
        setCardsDisabled(false);
        return;
      }
      if (!r.ok) return; // transient — next tick retries
      if (r.phase === "authorizing") {
        setStatus(t("waitAuth"));
      } else if (r.phase === "ready") {
        setStatus(t("ready"), "ok");
        stopPolling();
        setCardsDisabled(true);
      } else if (r.phase === "failed") {
        setStatus(t("failed") + (r.error || "unknown") + t("refreshHint"), "err");
        stopPolling();
        setCardsDisabled(false);
      } else {
        setStatus(t("selectPrompt"));
      }
    } finally {
      pollInFlight = false;
    }
  }

  function applyI18n() {
    document.title = t("title");
    document.documentElement.lang = lang === "zh" ? "zh-CN" : "en";
    var nodes = document.querySelectorAll("[data-i18n]");
    for (var i = 0; i < nodes.length; i++) {
      nodes[i].textContent = t(nodes[i].getAttribute("data-i18n"));
    }
    document.getElementById("lang").textContent = lang === "zh" ? "EN" : "中文";
  }
  document.getElementById("lang").addEventListener("click", function () {
    lang = lang === "zh" ? "en" : "zh";
    localStorage.setItem(LANG_STORAGE, lang);
    applyI18n();
    // Re-render the dynamic status line in the new language.
    pollOnce();
  });

  cards.forEach(function (card) {
    card.addEventListener("click", function () { start(card.getAttribute("data-provider")); });
  });
  applyI18n();
  // Resume an in-flight flow after a page reload (browser Back from the
  // authorize page lands here).
  pollOnce();
})();
</script>
