<script>
(function () {
  "use strict";
  var KEY_STORAGE = "zcode-login-key";
  var key = sessionStorage.getItem(KEY_STORAGE) || "";
  var statusEl = document.getElementById("status");
  var manualEl = document.getElementById("manual");
  var cards = Array.prototype.slice.call(document.querySelectorAll(".card"));
  var pollTimer = null;

  function esc(s) {
    return String(s).replace(/[&<>"']/g, function (c) {
      return { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c];
    });
  }
  function authHeaders() {
    return key ? { "authorization": "Bearer " + key } : {};
  }
  function setStatus(text, kind) {
    statusEl.textContent = text;
    statusEl.className = "status" + (kind ? " " + kind : "");
  }
  function setCardsDisabled(disabled) {
    cards.forEach(function (c) { c.disabled = disabled; });
  }
  // api(): shared fetch wrapper — on 401 (main-server proxyApiKey gate) prompt
  // once for the key, remember it in sessionStorage, and retry.
  async function api(path, opts) {
    var r = await fetch(path, opts);
    if (r.status === 401 && !key) {
      var k = prompt("此端点需要 Proxy API Key（config.yaml 中的 proxyApiKey）：");
      if (k && k.trim()) {
        key = k.trim();
        sessionStorage.setItem(KEY_STORAGE, key);
        var retryOpts = Object.assign({}, opts, { headers: Object.assign({}, (opts && opts.headers) || {}, authHeaders()) });
        r = await fetch(path, retryOpts);
      }
    }
    var parsed = null;
    try { parsed = await r.json(); } catch (e) { /* non-JSON */ }
    if (!parsed || typeof parsed.ok !== "boolean") {
      return { ok: false, error: r.status === 401 ? "unauthorized（API Key 不正确）" : "HTTP " + r.status };
    }
    return parsed;
  }

  async function start(provider) {
    setCardsDisabled(true);
    manualEl.classList.add("hidden");
    setStatus("正在获取授权地址…");
    var r = await api("/login/start", {
      method: "POST",
      headers: Object.assign({ "content-type": "application/json" }, authHeaders()),
      body: JSON.stringify({ provider: provider })
    });
    if (!r.ok) {
      setStatus("启动登录失败：" + (r.error || "unknown") + "（重试请刷新页面）", "err");
      setCardsDisabled(false);
      return;
    }
    // Jump to the OAuth login page in a new tab; the status page keeps
    // polling here. If the popup is blocked the manual link below is the way in.
    window.open(r.authorizeUrl, "_blank", "noopener");
    manualEl.innerHTML = '授权页：<a href="' + esc(r.authorizeUrl) + '" target="_blank" rel="noreferrer">点此打开</a>' +
      "（完成站点登录后回到本页查看结果；浏览器提示无法打开 zcode:// 链接属正常，忽略即可）";
    manualEl.classList.remove("hidden");
    setStatus("等待授权完成…");
    startPolling();
  }

  function stopPolling() {
    if (pollTimer !== null) { clearInterval(pollTimer); pollTimer = null; }
  }
  function startPolling() {
    stopPolling();
    pollOnce();
    pollTimer = setInterval(pollOnce, 2000);
  }
  var pollInFlight = false;
  async function pollOnce() {
    if (pollInFlight) return;
    pollInFlight = true;
    try {
      var r = await api("/login/status", { headers: authHeaders() });
      if (!r.ok) return; // transient — next tick retries
      if (r.phase === "authorizing") {
        setStatus("等待授权完成…（在浏览器里完成站点登录即可）");
      } else if (r.phase === "ready") {
        setStatus("✅ 登录成功，可以关闭此页", "ok");
        stopPolling();
        setCardsDisabled(true);
      } else if (r.phase === "failed") {
        setStatus("❌ 登录失败：" + (r.error || "unknown") + "（刷新页面可重试）", "err");
        stopPolling();
        setCardsDisabled(false);
      } else {
        setStatus("请选择上方站点开始登录");
      }
    } finally {
      pollInFlight = false;
    }
  }

  cards.forEach(function (card) {
    card.addEventListener("click", function () { start(card.getAttribute("data-provider")); });
  });
  // Resume an in-flight flow after a page reload (browser Back from the
  // authorize page lands here).
  pollOnce();
})();
</script>
