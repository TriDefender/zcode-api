    <script>
      const $ = (id) => document.getElementById(id);
      const STORAGE_KEY = "zcode.panel.token";
      let token = localStorage.getItem(STORAGE_KEY) || "";
      let lastCommand = null;
      let lastQuotaAt = 0;
      let logCursor = 0;
      let loggedIn = false;

      function render(obj) {
        $("raw").textContent = JSON.stringify(obj, null, 2);
      }

      function setConn(state, text) {
        $("dot").className = "dot " + state;
        $("conn").textContent = text;
      }

      async function cmd(payload) {
        const res = await fetch("/api/control", {
          method: "POST",
          headers: { "content-type": "application/json", authorization: "Bearer " + token },
          body: JSON.stringify(payload),
        });
        let body;
        try {
          body = await res.json();
        } catch (err) {
          body = { ok: false, error: "invalid_response" };
        }
        lastCommand = { request: payload, status: res.status, response: body };
        render(lastCommand);
        if (res.status === 401) setConn("bad", "unauthorized");
        return body;
      }

      function esc(value) {
        return String(value).replace(/[&<>"']/g, (ch) => (
          { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[ch]
        ));
      }

      /** Upstream timestamps arrive as seconds or milliseconds — accept both. */
      function fmtTs(value) {
        if (value == null) return "—";
        const ms = value < 1e12 ? value * 1000 : value;
        const date = new Date(ms);
        return Number.isNaN(date.getTime()) ? String(value) : date.toLocaleString();
      }

      async function refreshStatus() {
        if (!token) return;
        const r = await cmd({ cmd: "status" });
        if (!r.ok) {
          setConn("bad", r.error || "error");
          return;
        }
        setConn("ok", "connected");
        loggedIn = !!r.loggedIn;
        $("s-provider").textContent = r.provider;
        $("s-plan").textContent = r.plan;
        $("s-port").textContent = r.proxyPort ? "listening on " + r.proxyPort : "stopped";
        $("s-login").textContent = r.loggedIn ? "yes" : "no — use Login below";
        $("s-login").className = r.loggedIn ? "" : "danger";
        $("cfg-provider").value = r.provider;
        $("cfg-plan").value = r.plan;
        $("start").disabled = !!r.proxyPort;
        $("stop").disabled = !r.proxyPort;
      }

      function renderQuota(snap) {
        const parts = [];
        for (const b of snap.balances || []) {
          const remaining = b.remainingUnits;
          const total = b.totalUnits;
          const pct = total ? Math.min(100, Math.max(0, (remaining / total) * 100)) : null;
          parts.push(
            '<div class="row"><strong>' + esc(b.showName) + "</strong> <span class=\"muted\">" +
            esc(remaining) + " / " + esc(total) + " " + esc(b.unitType || "") +
            (b.expiresAt ? " · expires " + esc(fmtTs(b.expiresAt)) : "") +
            "</span></div>" +
            (pct === null ? "" : '<div class="bar"><i style="width:' + pct.toFixed(1) + '%"></i></div>'),
          );
        }
        const plan = snap.codingPlan;
        if (plan) {
          parts.push('<div class="row"><strong>' + esc(plan.level || "coding plan") + "</strong></div>");
          for (const limit of plan.limits || []) {
            // Upstream `number` is NOT a comparable total (live TIME_LIMIT row:
            // remaining=3894, number=1), so like the CLI/TUI this shows remaining
            // alone. `percentage` is the one self-consistent signal: when present
            // it is a used share (0-100), and the bar is the remaining share
            // derived from it — never an X/Y fabrication. Direction is not
            // re-labelled here because upstream owns its meaning.
            const remaining = limit.remaining;
            const pct = limit.percentage;
            const remainingFrac =
              typeof pct === "number" && isFinite(pct) && pct >= 0 && pct <= 100 ? 1 - pct / 100 : null;
            parts.push(
              '<div class="row"><span class="muted">' + esc(limit.type) + "</span> <strong>" +
              (typeof remaining === "number"
                ? esc(remaining.toLocaleString()) + (limit.unit ? " " + esc(limit.unit) : "") + " remaining"
                : "unavailable") +
              "</strong>" +
              (limit.nextResetTime ? ' <span class="muted">· resets ' + esc(fmtTs(limit.nextResetTime)) + "</span>" : "") +
              "</div>" +
              (remainingFrac === null
                ? ""
                : '<div class="bar"><i style="width:' + (remainingFrac * 100).toFixed(1) + '%"></i></div>'),
            );
          }
        } else {
          parts.push('<div class="muted">No coding-plan window for this account.</div>');
        }
        for (const err of snap.errors || []) {
          parts.push('<div class="danger">' + esc(err) + "</div>");
        }
        if (snap.jwt) {
          parts.push('<div class="muted">credential age: ' + esc(snap.jwt.ageHours.toFixed(1)) + " h</div>");
        }
        $("quota-body").innerHTML = parts.join("") || "Empty snapshot.";
      }

      async function loadQuota() {
        if (!token) return;
        const now = Date.now();
        if (now - lastQuotaAt < 60000) {
          $("quota-age").textContent = "please wait " + Math.ceil((60000 - (now - lastQuotaAt)) / 1000) + "s";
          return;
        }
        lastQuotaAt = now;
        $("quota-age").textContent = "loading…";
        const r = await cmd({ cmd: "quota" });
        if (r.ok && r.quota) {
          renderQuota(r.quota);
          $("quota-age").textContent = "updated " + new Date().toLocaleTimeString();
        } else {
          $("quota-age").textContent = "";
          $("quota-body").innerHTML = '<div class="danger">' + esc(r.error || "failed") + "</div>";
        }
      }

      async function pollLogs() {
        if (!token || !document.getElementById("logs-auto").checked) return;
        const r = await cmd({ cmd: "getLogs", since: logCursor });
        if (!r.ok) return;
        logCursor = r.nextSince;
        if (!r.lines.length) return;
        const box = $("logs");
        const stick = box.scrollTop + box.clientHeight >= box.scrollHeight - 24;
        box.textContent += r.lines.join("\n") + "\n";
        const lines = box.textContent.split("\n");
        if (lines.length > 500) box.textContent = lines.slice(lines.length - 500).join("\n");
        if (stick) box.scrollTop = box.scrollHeight;
      }

      async function startLogin(provider) {
        if (!token) return;
        $("login-body").innerHTML = '<span class="muted">requesting authorize URL…</span>';
        const r = await cmd({ cmd: "startOAuth", provider });
        if (!r.ok) {
          $("login-body").innerHTML = '<span class="danger">' + esc(r.error) + "</span>";
          return;
        }
        $("login-body").innerHTML =
          'Open this URL on any device: <a href="' + esc(r.authorizeUrl) + '" target="_blank" rel="noreferrer">' +
          esc(r.authorizeUrl) + "</a>";
        // The server finishes the flow by itself (polling login, no local
        // callback needed), so poll briefly to flip the status card.
        let tries = 0;
        const timer = setInterval(async () => {
          tries += 1;
          await refreshStatus();
          if (loggedIn || tries >= 100) clearInterval(timer);
        }, 3000);
      }

      async function guard(label, payload) {
        if (!token) return;
        const r = await cmd(payload);
        if (!r.ok) $("cfg-note").textContent = label + ": " + r.error;
        else $("cfg-note").textContent = label + ": ok";
        await refreshStatus();
      }

      $("connect").onclick = async () => {
        token = $("token").value.trim();
        localStorage.setItem(STORAGE_KEY, token);
        logCursor = 0;
        $("logs").textContent = "";
        await refreshStatus();
      };
      $("refresh").onclick = () => refreshStatus();
      $("quota").onclick = () => loadQuota();
      $("start").onclick = () => guard("start", { cmd: "startProxy" });
      $("stop").onclick = () => guard("stop", { cmd: "stopProxy" });
      $("apply").onclick = () =>
        guard("apply", { cmd: "setConfig", provider: $("cfg-provider").value, plan: $("cfg-plan").value });
      $("login").onclick = () => startLogin("zai");
      $("login-cn").onclick = () => startLogin("zai_cn");
      $("logout").onclick = () => {
        if (
          confirm("Log out and delete the stored credential? The proxy stops using it immediately and is stopped if it is running.")
        )
          guard("logout", { cmd: "logout" });
      };
      $("logs-clear").onclick = () => { $("logs").textContent = ""; };

      if (token) {
        $("token").value = token;
        refreshStatus();
        loadQuota();
      } else {
        setConn("bad", "enter token to connect");
      }
      setInterval(refreshStatus, 5000);
      setInterval(pollLogs, 2000);
      // No quota polling timer on purpose: the billing gateway rate-limits
      // frequent queries (same reason the TUI refreshes on keypress only).
    </script>
