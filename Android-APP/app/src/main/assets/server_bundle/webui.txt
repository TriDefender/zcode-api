      function renderMarkdown(text) {
        if (typeof marked === "undefined") return escapeHtml(text);
        try {
          marked.setOptions({ gfm: true, breaks: true, headerIds: false, mangle: false });
          var html = marked.parse(text);
          if (typeof DOMPurify !== "undefined") html = DOMPurify.sanitize(html);
          return html;

      }
      function escapeHtml(s) {
        return s.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");

      // Tiny script appended to rendered HTML so the sandboxed iframe can tell
      // its parent how tall the content is (auto-height). Split "<script" so the
      // host parser never sees a real end-tag inside this string.
      var RESIZE_SCRIPT = '<' + 'script>(function(){function r(){try{parent.postMessage({__zcpPreview:true,h:document.documentElement.scrollHeight},"*")}catch(e){}}window.addEventListener("load",r);setTimeout(r,60);setTimeout(r,400);setTimeout(r,1200);})();<' + '/script>';
      var previewListenerAdded = false;
      function ensurePreviewListener() {
        if (previewListenerAdded) return;
        previewListenerAdded = true;
        window.addEventListener("message", function (ev) {
          var d = ev.data;
          if (!d || d.__zcpPreview !== true || typeof d.h !== "number") return;
          var frames = document.querySelectorAll("iframe.html-preview");
          for (var i = 0; i < frames.length; i++) {
            if (frames[i].contentWindow === ev.source) {
              var h = Math.max(120, Math.min(2000, d.h | 0));
              frames[i].style.height = h + "px";
              break;
            }
          }
        });
      }
      function toggleHtmlPreview(pre, code, btn, label, raw) {
        var frame = pre.querySelector("iframe.html-preview");

          // null origin that cannot touch the parent page's DOM/storage.

      // =========================================================
      // MCP — client-managed (browser-direct, optional CORS proxy).
      // The webui connects to each server, lists its tools, injects them as
      // function tools, and auto-executes tool calls the model returns.
      // =========================================================
      var mcpState = {};          // url -> {sessionId, tools[], connected, error, status}
      var mcpToolMap = {};        // tool name -> {url, name}
      var mcpRpcId = 0;
      function nextRpcId() { return ++mcpRpcId; }

      function applyCorsProxy(url) {
        var cp = (state.settings.mcpCorsProxy || "").trim();
        if (!cp) return url;
        if (cp.indexOf("{url}") >= 0) return cp.replace("{url}", encodeURIComponent(url));
        return cp + encodeURIComponent(url);
      }
      // POST one JSON-RPC message. Resolves {status, contentType, sessionId, body}.
      // Auth: sent only as Authorization: Bearer (KoboldAI parity). If a server
      // needs the key in the URL, embed it directly in the server URL line.
      function mcpFetch(server, rpc) {
        var st = mcpState[server.url] || (mcpState[server.url] = {});
        var headers = { "content-type": "application/json", "accept": "application/json, text/event-stream" };
        if (st.sessionId) headers["mcp-session-id"] = st.sessionId;
        if (server.authKey) headers["authorization"] = "Bearer " + server.authKey;
        return fetch(applyCorsProxy(server.url), {
          method: "POST", headers: headers, body: JSON.stringify(rpc),
        }).then(function (r) {
          return r.text().then(function (t) {
            return {
              ok: r.ok, status: r.status,
              contentType: r.headers.get("content-type") || "",
              sessionId: r.headers.get("mcp-session-id") || "",
              body: t,
            };
          });
        });
      }
      // streamable-http servers may answer with SSE (data: lines) or plain JSON.
      function parseMcpResponse(resp) {
        var body = resp.body || "";
        var looksSSE = resp.contentType.indexOf("text/event-stream") >= 0 || body.indexOf("data:") === 0;
        if (looksSSE) {
          var parsed = null;
          body.split("\n").forEach(function (ln) {
            if (ln.indexOf("data:") === 0) {
              var d = ln.slice(5).trim();
              if (d) { try { parsed = JSON.parse(d); } catch (e) {} }
            }
          });
          return parsed;
        }
        try { return JSON.parse(body); } catch (e) { return null; }
      }
      function mcpCall(server, method, params) {
        var rpc = { jsonrpc: "2.0", id: nextRpcId(), method: method };
        if (params) rpc.params = params;
        return mcpFetch(server, rpc).then(function (resp) {
          var st = mcpState[server.url];
          if (resp.sessionId && st && !st.sessionId) st.sessionId = resp.sessionId;
          var json = parseMcpResponse(resp);
          if (!json) throw new Error("MCP " + method + " on " + server.url + " returned no JSON (HTTP " + resp.status + ")");
          if (json.error) throw new Error("MCP error" + (json.error.code != null ? " (" + json.error.code + ")" : "") + ": " + (json.error.message || JSON.stringify(json.error)));
          return json;
        });
      }
      // initialize + tools/list for one server. Resolves to tools[].
      function mcpConnect(server) {
        return mcpCall(server, "initialize", {
          protocolVersion: "2024-11-05", capabilities: {},
          clientInfo: { name: "zcode-proxy-webui", version: "1.0" },
        }).then(function () {
          // required notification before tools/list on stateful servers
          return mcpFetch(server, { jsonrpc: "2.0", method: "notifications/initialized" });
        }).then(function () {
          return mcpCall(server, "tools/list", {});
        }).then(function (j) {
          return (j.result && j.result.tools) || [];
        });
      }
      // Build function-tool entries from every connected server + refresh the
      // tool name -> {url, name} routing map. No tools when MCP is disabled.
      function gatherMcpTools() {
        var out = []; mcpToolMap = {};
        if (!state.settings.mcpEnabled) return out;
        state.settings.mcpServers.forEach(function (s) {
          var st = mcpState[s.url];
          if (st && st.connected && st.tools) {
            st.tools.forEach(function (t) {
              if (mcpToolMap[t.name]) return; // dedupe across servers (first wins)
              mcpToolMap[t.name] = { url: s.url, name: t.name };
              out.push({
                type: "function",
                function: {
                  name: t.name,
                  description: (t.description || t.name).slice(0, 1024),
                  parameters: t.inputSchema || { type: "object", properties: {} },
                },
              });
            });
          }
        });
        return out;
      }
      // Execute one model tool_call against its MCP server. Resolves to text.
      function execMcpTool(tc) {
        var map = mcpToolMap[tc.name];
        if (!map) return Promise.reject(new Error("Unknown tool: " + tc.name));
        var server = state.settings.mcpServers.find(function (s) { return s.url === map.url; });
        if (!server) return Promise.reject(new Error("MCP server not configured: " + map.url));
        var args = {};
        try { args = JSON.parse(tc.arguments || "{}"); } catch (e) { args = {}; }
        return mcpCall(server, "tools/call", { name: map.name, arguments: args }).then(function (j) {
          var res = j.result;
          if (!res) return "(no result)";
          if (Array.isArray(res.content)) {
            return res.content.map(function (c) { return c.type === "text" ? c.text : JSON.stringify(c); }).join("\n");
          }
          return JSON.stringify(res);
        });
      }

      // Parse the MCP textarea (KoboldAI format: "URL" or "URL,key" per line).
      function parseServersFromTextarea() {
        return $("s-mcp").value.split("\n").map(function (ln) {
          var line = ln.trim();
          if (!line) return null;
          var parts = line.split(",");
          var url = (parts[0] || "").trim();
          var key = (parts[1] || "").trim();
          if (!url) return null;
          return { url: url, authKey: key };
        }).filter(function (s) { return s.url; });
      }
      function connectAllMcp() {
        var servers = parseServersFromTextarea();
        state.settings.mcpServers = servers;
        state.settings.mcpCorsProxy = $("s-cors").value.trim();
        state.settings.mcpEnabled = $("s-mcpenable").checked;
        saveState();
        mcpState = {};              // fresh sessions each connect
        renderMcpStatus(servers);
        var chain = Promise.resolve();
        servers.forEach(function (s) {
          chain = chain.then(function () {
            var st = mcpState[s.url] = { connected: false, tools: [], sessionId: "", error: "", status: "connecting" };
            renderMcpStatus(servers);
            return mcpConnect(s).then(function (tools) {
              st.connected = true; st.tools = tools || []; st.error = ""; st.status = "ok";
            }, function (err) {
              st.connected = false; st.tools = []; st.error = String((err && err.message) || err); st.status = "error";
            }).then(function () { renderMcpStatus(servers); });
          });
        });

      function send() {
        if (busy) return;
        var text = inputEl.value.trim();
        if (!text && pendingImages.length === 0) return;
        // Block image input for non-vision models with a clear message.
        if (pendingImages.length && !isVisionModel(state.settings.model)) {
          toast("This model does not support image input. Select a model marked as image-capable.");
          return;
        }
        var c = activeConv(); if (!c) { newConv(); c = activeConv(); }
        var images = pendingImages.map(function (p) { return p.dataUrl; });
        c.messages.push({ role: "user", content: text, images: images.length ? images : undefined });
        if (c.messages.filter(function (m) { return m.role === "user"; }).length === 1 && text) {
          c.title = text.slice(0, 40);
        }
        inputEl.value = ""; autoGrow(); pendingImages = []; renderAttachPreview(); renderSidebar();
        renderMessages();
        runCompletion(c);
      }

      function regenerate() {
        if (busy) return;
        var c = activeConv(); if (!c) return;
        while (c.messages.length && c.messages[c.messages.length - 1].role === "assistant") c.messages.pop();
        renderMessages();
        runCompletion(c);
      }

      function runCompletion(conv) {
        setBusy(true);
        runToolLoop(conv, 0);
      }

      // One assistant turn per call; recurses while the model requests tools.
      function runToolLoop(conv, depth) {
        if (depth > 8) {
          toast("Stopped after 8 tool rounds.");
          finalizeLoop();
          return;
        }
        var assistant = { role: "assistant", content: "", reasoning: "", model: conv.model || state.settings.model, truncated: false };
        conv.messages.push(assistant);
        renderMessages();

        streamOnce(conv, assistant).then(function (toolCalls) {
          if (toolCalls && toolCalls.length) {
            assistant.toolCalls = toolCalls;
            saveState(); renderMessages();
            // Auto-execute every tool call against its MCP server, sequentially.
            var seq = Promise.resolve();
            toolCalls.forEach(function (tc) {
              seq = seq.then(function () {
                return execMcpTool(tc).then(function (result) {
                  conv.messages.push({ role: "tool", toolCallId: tc.id, content: result });
                }, function (err) {
                  conv.messages.push({ role: "tool", toolCallId: tc.id, content: "Error: " + String((err && err.message) || err) });
                });
              });
            });
            seq.then(function () {
              saveState(); renderMessages(); scrollBottom(true);
              runToolLoop(conv, depth + 1);   // feed results back to the model
            });
          } else {
            finalizeLoop();
          }
        });

        function finalizeLoop() {
          setBusy(false);
          abortCtrl = null;
          saveState();
          renderSidebar();
        }
      }

      // Stream one completion. Resolves to an array of accumulated tool calls
      // (empty/null when the model just replied with text).
      function streamOnce(conv, assistant) {
        return new Promise(function (resolve) {
          abortCtrl = new AbortController();
          var toolAccum = {};     // index -> {id, name, arguments}
          var finished = false;

          fetch("/v1/chat/completions", {
            method: "POST",
            headers: {
              "content-type": "application/json",
              "authorization": state.settings.apiKey ? "Bearer " + state.settings.apiKey : "",
            },
            body: JSON.stringify(buildBody(conv)),
            signal: abortCtrl.signal,
          }).then(function (r) {
            if (!r.ok || !r.body) {
              return r.text().then(function (t) {
                assistant.content = "**Request failed (" + r.status + ").**\n\n```\n" + (t || r.statusText).slice(0, 800) + "\n```";
                updateStreamingDom(assistant, true);
                resolve(null);
                throw new Error("__stop__");
              });
            }
            var reader = r.body.getReader();
            var dec = new TextDecoder();
            var buf = "";
            var lastRender = 0;
            function pump() {
              return reader.read().then(function (chunk) {
                if (chunk.done) { return; }
                buf += dec.decode(chunk.value, { stream: true });
                var lines = buf.split("\n");
                buf = lines.pop();
                for (var i = 0; i < lines.length; i++) {
                  var line = lines[i].trim();
                  if (!line || line.indexOf("data:") !== 0) continue;
                  var data = line.slice(5).trim();
                  if (data === "[DONE]") { finished = true; break; }
                  var json; try { json = JSON.parse(data); } catch (e) { continue; }
                  var choice = json.choices && json.choices[0];
                  if (!choice) continue;
                  var d = choice.delta || {};
                  if (d.reasoning_content) assistant.reasoning += d.reasoning_content;
                  if (d.content) assistant.content += d.content;
                  if (d.tool_calls) {
                    d.tool_calls.forEach(function (tc) {
                      var idx = tc.index || 0;
                      var slot = toolAccum[idx] = toolAccum[idx] || { id: "", name: "", arguments: "" };
                      if (tc.id) slot.id = tc.id;
                      if (tc.function) {
                        if (tc.function.name) slot.name = tc.function.name;
                        if (tc.function.arguments) slot.arguments += tc.function.arguments;
                      }
                    });
                  }
                  if (choice.finish_reason === "length") assistant.truncated = true;
                }
                var now = Date.now();
                if (now - lastRender > 33) { lastRender = now; updateStreamingDom(assistant, false); }
                if (finished) { return; }
                return pump();
              });
            }
            return pump().then(function () {
              updateStreamingDom(assistant, true);
              var tcs = Object.keys(toolAccum).map(function (k) { return toolAccum[k]; }).filter(function (t) { return t.id && t.name; });
              if (!assistant.content && !assistant.reasoning && !tcs.length) {
                assistant.content = "_(empty response)_"; updateStreamingDom(assistant, true);
              }
              resolve(tcs.length ? tcs : null);
            });
          }).catch(function (e) {
            if (e && e.message === "__stop__") return;
            if (e && e.name === "AbortError") {
              assistant.content += "\n\n_\u2026stopped._"; updateStreamingDom(assistant, true);
            } else {
              assistant.content = "**Error:** " + escapeHtml(String((e && e.message) || e));
              updateStreamingDom(assistant, true);
            }
            resolve(null);
          });
        });
      }

      function updateStreamingDom(assistant, done) {
        var last = msgList.lastChild;
        if (!last || !last.classList || !last.classList.contains("msg")) { renderMessages(); last = msgList.lastChild; }
        if (!last) return;
        var body = last.querySelector(".content");
        if (!body) return;
        var think = body.querySelector(".think");
        if (assistant.reasoning) {
          if (!think) {
            think = document.createElement("details"); think.className = "think"; think.open = true;
            var sum = document.createElement("summary");
            var tb = document.createElement("div"); tb.className = "think-body";
            think.appendChild(sum); think.appendChild(tb);
            body.insertBefore(think, body.firstChild);
          }
          think.querySelector(".think-body").textContent = assistant.reasoning;
          think.querySelector("summary").textContent = done ? "Thought process" : "Thinking\u2026";
          if (done) think.open = false;
        }
        if (assistant.content) {
          body.innerHTML = renderMarkdown(assistant.content);
          enhanceCode(body);
          if (think) body.insertBefore(think, body.firstChild); // keep thinking on top
        }
        if (done) {
          var oldMeta = last.querySelector(".msg-meta"); if (oldMeta) oldMeta.remove();
          last.appendChild(meta(assistant, true));
        }
        scrollBottom();
      }
